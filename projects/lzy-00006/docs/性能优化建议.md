# 实时语音对讲模块 性能优化建议

## 1. 概述

本文档提供了实时语音对讲模块的性能优化建议，旨在进一步提升语音质量、降低延迟、减少资源占用，以及提高系统可扩展性。

2. **优先选择 Opus 编解码器**
```javascript
// 强制使用 Opus 编解码器
function preferOpus(sdp) {
    const sdpLines = sdp.split('\r\n');
    const mLineIndex = sdpLines.findIndex(line => line.startsWith('m=audio'));
    
    if (mLineIndex === -1) return sdp;
    
    const opusPayloadType = sdpLines
        .filter(line => line.includes('opus/48000'))
        .map(line => line.match(/a=rtpmap:(\d+) opus\/48000/i))
        .filter(match => match)
        .map(match => match[1])[0];
    
    if (opusPayloadType) {
        const mLine = sdpLines[mLineIndex];
        const parts = mLine.split(' ');
        const payloads = parts.slice(3).filter(p => p !== opusPayloadType);
        parts.splice(3, parts.length - 3, opusPayloadType, ...payloads);
        sdpLines[mLineIndex] = parts.join(' ');
    }
    
    return sdpLines.join('\r\n');
}
```

```javascript
// 根据环境噪声动态调整阈值
class AdaptiveVAD {
    constructor() {
        this.noiseFloor = -60;  // 初始噪声底
        this.thresholdOffset = 10;  // 阈值偏移
        this.smoothingFactor = 0.95;
    }

    updateNoiseFloor(db) {
        if (db < this.noiseFloor + 5) {
            this.noiseFloor = this.smoothingFactor * this.noiseFloor + 
                             (1 - this.smoothingFactor) * db;
        }
    }

    isSpeech(db) {
        return db > this.noiseFloor + this.thresholdOffset;
    }
}
```

### 3.2 自适应码率控制（ABS）优化

**当前实现**：基础的基于 RTT 和丢包的码率调整

**优化建议**：

1. **更精细的码率控制算法**
```javascript
class AdvancedBitrateController {
    constructor() {
        this.currentBitrate = 64000;
        this.targetBitrate = 64000;
        this.minBitrate = 16000;
        this.maxBitrate = 128000;
        
        // PID 控制器参数
        this.kp = 0.3;  // 比例系数
        this.ki = 0.1;  // 积分系数
        this.kd = 0.05; // 微分系数
        
        this.rttHistory = [];
        this.integral = 0;
        this.prevError = 0;
    }

    update(stats) {
        const { avgRtt, totalPacketsLost, peerCount } = stats;
        const targetRtt = 150;  // 目标延迟 150ms
        
        // PID 控制
        const error = targetRtt - avgRtt;
        this.integral += error;
        const derivative = error - this.prevError;
        
        let adjustment = this.kp * error + this.ki * this.integral + this.kd * derivative;
        
        // 丢包惩罚
        if (totalPacketsLost > 5) {
            adjustment *= 0.5;
        }
        
        // 人数调整
        const peerFactor = Math.max(0.5, 1 - (peerCount - 2) * 0.1);
        this.targetBitrate = Math.min(
            this.maxBitrate * peerFactor,
            Math.max(this.minBitrate, this.currentBitrate + adjustment)
        );
        
        // 平滑过渡
        this.currentBitrate = 0.8 * this.currentBitrate + 0.2 * this.targetBitrate;
        
        this.prevError = error;
        return Math.round(this.currentBitrate);
    }
}
```

2. **带宽探测**
```javascript
// 定期探测可用带宽
async function probeBandwidth(peerConnection) {
    const sender = peerConnection.getSenders()[0];
    if (!sender) return;
    
    const params = sender.getParameters();
    const originalBitrate = params.encodings[0].maxBitrate;

            // 恢复原码率

            await sender.setParameters(params);
        }

}
```

```javascript
class PeerConnectionPool {
    constructor(maxSize = 10) {
        this.pool = [];
        this.maxSize = maxSize;
    }

    acquire(config) {
        if (this.pool.length > 0) {
            return this.pool.pop();
        }
        return new RTCPeerConnection(config);
    }

    release(pc) {
        if (this.pool.length < this.maxSize) {
            // 重置连接状态但不关闭
            pc.ontrack = null;
            pc.onicecandidate = null;
            pc.oniceconnectionstatechange = null;
            this.pool.push(pc);
        } else {
            pc.close();
        }
    }
}
```

```javascript
// 限制每个房间的语音会话数量
const MAX_ROOMS = 1000;
const voiceSessions = new Map();

// LRU 清理策略
function cleanupOldSessions() {
    const now = Date.now();
    const maxAge = 24 * 60 * 60 * 1000;  // 24小时
    
    for (const [roomCode, session] of voiceSessions) {
        if (now - session.createdAt > maxAge) {
            voiceSessions.delete(roomCode);
        }
    }
    
    // 限制总数量
    if (voiceSessions.size > MAX_ROOMS) {
        const entries = Array.from(voiceSessions.entries());
        entries.sort((a, b) => a[1].createdAt - b[1].createdAt);
        const toDelete = entries.slice(0, voiceSessions.size - MAX_ROOMS);
        toDelete.forEach(([roomCode]) => voiceSessions.delete(roomCode));
    }
}

setInterval(cleanupOldSessions, 60 * 60 * 1000);  // 每小时清理一次
```

## 6. 监控与告警

### 6.1 语音质量监控

1. **客户端质量指标收集**
```javascript
// 定期上报质量指标
class VoiceQualityMonitor {
    constructor(voiceManager) {
        this.voiceManager = voiceManager;
        this.qualityHistory = [];
    }

    start() {
        setInterval(async () => {
            const stats = await this.voiceManager._getNetworkStats();
            if (stats) {
                this.qualityHistory.push({
                    timestamp: Date.now(),
                    ...stats
                });
                
                // 保留最近 10 分钟数据
                const tenMinutesAgo = Date.now() - 10 * 60 * 1000;
                this.qualityHistory = this.qualityHistory.filter(
                    q => q.timestamp > tenMinutesAgo
                );
                
                // 检测质量问题
                this._detectIssues();
            }
        }, 5000);
    }

    _detectIssues() {
        const recent = this.qualityHistory.slice(-6);  // 最近 30 秒
        
        if (recent.length < 3) return;
        
        const avgRtt = recent.reduce((sum, q) => sum + q.avgRtt, 0) / recent.length;
        const avgPacketLoss = recent.reduce((sum, q) => sum + q.totalPacketsLost, 0) / recent.length;
        
        if (avgRtt > 300) {
            console.warn('High latency detected:', avgRtt);
            // 触发降级策略
        }
        
        if (avgPacketLoss > 20) {
            console.warn('High packet loss detected:', avgPacketLoss);
            // 通知用户网络问题
        }
    }
}
```

## 7. 可扩展性优化

### 7.1 从 Mesh 到 SFU 架构

当需要支持 >10 人通话时，建议迁移到 SFU 架构：

**架构对比**：

| 特性 | Mesh (当前) | SFU | MCU |
|------|------------|-----|-----|
| 最大人数 | ~10人 | 100+ | 1000+ |
| 服务器负载 | 低（仅信令） | 中（媒体转发） | 高（媒体混码） |
| 客户端带宽 | O(n) 线性增长 | O(1) 恒定 | O(1) 恒定 |
| 客户端CPU | 高（n-1个编码） | 低（1个编码） | 低（1个编码） |
| 隐私性 | 端到端加密 | 服务器可见 | 服务器可见 |
| 开发复杂度 | 低 | 中 | 高 |

**推荐 SFU 方案**：
1. **Mediasoup** - 高性能、C++ 内核、Node.js 绑定
2. **LiveKit** - 现代化、Go 语言、全栈 SDK
3. **Janus** - 成熟稳定、插件式架构

### 7.2 多地域部署

```javascript
// 基于地理位置选择最近的边缘节点
async function getNearestEdgeServer() {
    const regions = [
        { id: 'us-east', url: 'https://us-east.voice.example.com' },
        { id: 'us-west', url: 'https://us-west.voice.example.com' },
        { id: 'eu-west', url: 'https://eu-west.voice.example.com' },
        { id: 'ap-east', url: 'https://ap-east.voice.example.com' }
    ];
    
    // 并行 ping 所有节点
    const results = await Promise.all(
        regions.map(async region => {
            const start = Date.now();
            try {
                await fetch(region.url + '/ping');
                return { region, latency: Date.now() - start };
            } catch {
                return { region, latency: Infinity };
            }
        })
    );
    
    // 选择延迟最低的节点
    results.sort((a, b) => a.latency - b.latency);
    return results[0].region;
}
```
